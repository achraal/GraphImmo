// RECRÉER PROCHE_DE < 5km
MATCH (p1:Propriete), (p2:Propriete)
WHERE id(p1) < id(p2) 
  AND p1.geo IS NOT NULL AND size(p1.geo) = 2 
  AND p2.geo IS NOT NULL AND size(p2.geo) = 2
WITH p1, p2, 
  round(point.distance(
    point({latitude: p1.geo[0], longitude: p1.geo[1]}),
    point({latitude: p2.geo[0], longitude: p2.geo[1]})
  ) / 1000, 2) AS distance_km
WHERE distance_km <= 2.0  // ← 5km au lieu de 10
CREATE (p1)-[:PROCHE_DE {distanceKm: distance_km}]->(p2)
RETURN count(*) AS nb_proche_de_5km;


// Version SIMPLE : équipements IDENTIQUES exacts
MATCH (p1:Propriete)-[:POSSEDE_AMENAGEMENT]->(a1),
      (p2:Propriete)-[:POSSEDE_AMENAGEMENT]->(a2:Amenagement)
WHERE id(p1) < id(p2) 
  AND a1.type = a2.type  // MÊME type équipement
  AND COUNT { (p1)-[:POSSEDE_AMENAGEMENT]->() } >= 1 
  AND COUNT { (p2)-[:POSSEDE_AMENAGEMENT]->() } >= 1
WITH p1, p2, 
  // Score = proportion équipements communs
  toFloat(count(DISTINCT a1)) / 
  (COUNT { (p1)-[:POSSEDE_AMENAGEMENT]->() } + COUNT { (p2)-[:POSSEDE_AMENAGEMENT]->() }) * 2 AS score
WHERE score >= 0.5
MERGE (p1)-[s:SIMILAIRE_PAR_AMENAGEMENT]->(p2)
SET s.scoreAmenagement = round(score, 3)
RETURN count(DISTINCT s) AS nb_similaire_amenagement;

MATCH (p1:Propriete {geo: null}) DETACH DELETE p1;  // Nettoyer geo null

WHERE id(p1) < id(p2) AND size(p1.geo)=2 AND size(p2.geo)=2
WITH p1, p2, round(point.distance(
  point({latitude: p1.geo[0], longitude: p1.geo[1]}),
  point({latitude: p2.geo[0], longitude: p2.geo[1]})
)/1000, 2) AS dist
WHERE dist <= 2
CREATE (p1)-[:PROCHE_DE {distanceKm: dist}]->(p2);

RETURN 'PROCHE_DE OK (' + count(*) + ' relations)';

// ============ 2. SIMILAIRE_PAR_AMENAGEMENT ============
MATCH (p1:Propriete)-[r:SIMILAIRE_PAR_AMENAGEMENT]->(p2) DELETE r;

MATCH (p1:Propriete)-[:POSSEDE_AMENAGEMENT]->(a1:Amenagement),
      (p2:Propriete)-[:POSSEDE_AMENAGEMENT]->(a2:Amenagement)
WHERE id(p1) < id(p2)
WITH p1, p2, 
  size([t IN apoc.coll.intersection(
    collect(DISTINCT a1.type), collect(DISTINCT a2.type)
  ) | t]) * 1.0 / 
  size(apoc.coll.union(
    collect(DISTINCT a1.type), collect(DISTINCT a2.type)
  )) AS score
WHERE score >= 0.5
CREATE (p1)-[:SIMILAIRE_PAR_AMENAGEMENT {scoreAmenagement: round(score, 3)}]->(p2);

RETURN 'SIMILAIRE_PAR_AMENAGEMENT OK';
