query ProprieteComplete($id: String!) {
  proprieteDetail(id: $id) {
    # Propriété principale
    id
    titre
    prix
    adresse
    description
    surface
    chambres
    lat
    lon

    # Photos (avec angle)
    photos {
      url
      angle
    }

    # Aménagements complets
    amenagements {
      id
      type
      qualite
    }

    # Localisation complète
    localisation {
      id
      commune
      region
      prixM2
      securite
    }

    # Agence complète
    agence {
      id
      nom
      adresse
      email
      telephone
    }

    # Champs des relations
    dateDepot
    prixM2Local
  }
}
