' =====================
' Enumerations
' =====================
enum RoleUtilisateur {
  ACHETEUR
  LOCATAIRE
  VENDEUR
  AGENT_IMMOBILIER
}

enum StatutTransaction {
  EN_COURS
  TERMINEE
  ANNULEE
}

enum StatutDemandeVisite {
  EN_ATTENTE
  ACCEPTEE
  REFUSEE
}

enum QualitePhoto {
  SD
  HD
}

' =====================
' Classes (nœuds Neo4j)
' =====================
class User {
  +id: String
  +nom: String
  +prenom: String
  +email: String
  +budget: Float
  +surfaceMin: Float
  +villePrefs: String
  +preferencesLocation: String
  +role: RoleUtilisateur
}

class Propriete {
  +id: String
  +titre: String
  +description: String
  +prix: Float
  +surface: Float
  +chambres: Int
  +adresse: String
  +geoLat: Float
  +geoLon: Float
}

class Agence {
  +id: String
  +nom: String
  +telephone: String
  +email: String
  +adresse: String
}

class Localisation {
  +latitude: Float
  +longitude: Float
  +commune: String
  +region: String
  +prixM2: Float
  +securite: String
}

class Transaction {
  +id: String
  +date: DateTime
  +montant: Float
  +statut: StatutTransaction
}

class Amenagement {
  +type: String
  +qualite: String
}

class Photo {
  +url: String
  +angle: String
  +dateUpload: DateTime
  +qualite: QualitePhoto
}

class HistoriqueVisite {
  +dateVisite: DateTime
  +notes: String
  +prixPropose: Float
}

' =====================
' Relations Neo4j (avec propriétés)
' =====================
class Favori <<relationship>> {
  +dateAjout: DateTime
  +note: Int
}

class VisiteRel <<relationship>> {
  +date: DateTime
  +impression: String
}

class Gestion <<relationship>> {
  +dateDepot: DateTime
}

class ProcheDe <<relationship>> {
  +distanceKm: Float
}

class SimilaireA <<relationship>> {
  +scoreSimilarite: Float
}

class InteressePar <<relationship>> {
  +priorite: Int
}

class RecommandeePour <<relationship>> {
  +score: Float
}

class DansQuartier <<relationship>> {
  +prixM2Local: Float
}

class NotificationRel <<relationship>> {
  +type: String
  +date: DateTime
  +lu: Boolean
  +message: String
  +proprieteId: String
}

class DemandeVisite <<relationship>> {
  +dateSouhaitee: Date
  +heure: String
  +statut: StatutDemandeVisite
  +dateCreation: DateTime
  +message: String
}

class SimilaireParAmenagement <<relationship>> {
  +scoreAmenagement: Float
}
